        <!-- Live Tips / Prediction Section Header -->

            <h2 class="text-xl font-semibold">Today's Featured Predictions</h2>
            <button onclick="fetchMatches()" class="bg-green-500 hover:bg-green-600 text-gray-950 font-bold py-2 px-4 rounded text-xs transition duration-200">
                🔄 Refresh Odds
            </button>
        </div>

        <!-- Dynamic Match Container (JavaScript will fill this) -->
        <div id="match-container" class="space-y-4">
            <div class="text-center py-8 text-gray-500">Loading live fixtures and analyzing probabilities...</div>

    <!-- Frontend logic to fetch and display prediction algorithms -->
    <script>
        // Using a free, no-auth testing endpoint from API-Football for live fixtures
        const API_URL = "https://api-sports.io";

        async function fetchMatches() {
            const container = document.getElementById('match-container');
            
            try {
                // Fetching actual data straight from the visitor's browser (0 cost to you)
                const response = await fetch(API_URL);
                const data = await response.json();
                
                // If no live matches or server error, fallback to demo mockup data so page is never blank
                if (!data.response || data.response.length === 0) {
                    showMockData();
                    return;
                }

                container.innerHTML = ''; // Clear loading screen

                // Loop through real incoming API match arrays (limiting to 3 for clean view)
                data.response.slice(0, 3).forEach(item => {
                    const homeTeam = item.teams.home.name;
                    const awayTeam = item.teams.away.name;
                    const league = item.league.name;
                    
                    // Simple static UI math script acting as your prediction "AI engine"
                    const homeProb = Math.floor(Math.random() * (65 - 35 + 1)) + 35;
                    const awayProb = 100 - homeProb;
                    const prediction = homeProb > awayProb ? `${homeTeam} Win/Draw` : `${awayTeam} Win/Draw`;

                    container.appendChild(createMatchCard(league, homeTeam, awayTeam, homeProb, awayProb, prediction));
                });

            } catch (error) {
                console.log("API limit reached or blocked by CORS. Loading demo layout.");
                showMockData();
            }
        }

        // Generates clean HTML blocks dynamically
        function createMatchCard(league, home, away, homeProb, awayProb, prediction) {
            const div = document.createElement('div');
            div.className = "bg-gray-950 border border-gray-800 rounded-xl p-5 shadow-lg flex flex-col md:flex-row justify-between items-center gap-4";
            div.innerHTML = `
                <div class="flex-1 w-full text-center md:text-left">
                    <span class="text-xs font-semibold bg-gray-800 text-green-400 px-2.5 py-1 rounded-full uppercase tracking-wider">${league}</span>
                    <div class="text-lg font-bold mt-3 flex items-center justify-center md:justify-start gap-4">
                        <span>${home}</span> <span class="text-xs text-gray-500">vs</span> <span>${away}</span>
                    </div>
                </div>
                <div class="flex items-center gap-6 w-full md:w-auto justify-between md:justify-end border-t md:border-t-0 border-gray-800 pt-3 md:pt-0">
                    <div class="text-center">
                        <div class="text-xs text-gray-400 uppercase font-mono">Probabilities</div>
                        <div class="text-sm font-semibold mt-1 text-blue-400">${homeProb}% H | ${awayProb}% A</div>
                    </div>
                    <div class="bg-green-500/10 border border-green-500/30 px-4 py-2 rounded-lg text-center min-w-[120px]">
                        <div class="text-[10px] text-green-400 uppercase font-bold tracking-wider">Prediction</div>
                        <div class="text-sm font-bold text-green-300 mt-0.5">${prediction}</div>
                    </div>
                </div>
            `;
            return div;
        }

        // Fallback mock layout just in case the public server demo key hits an hourly cap
        function showMockData() {
            const container = document.getElementById('match-container');
            container.innerHTML = '';
            
            const mockMatches = [
                { league: "English Premier League", home: "Arsenal", away: "Chelsea", hP: 54, aP: 46, pred: "Arsenal Win" },
                { league: "La Liga", home: "Real Madrid", away: "Barcelona", hP: 48, aP: 52, pred: "Over 2.5 Goals" },
                { league: "UEFA Champions League", home: "Man City", away: "Bayern Munich", hP: 61, aP: 39, pred: "Man City Win" }
            ];

            mockMatches.forEach(m => {
                container.appendChild(createMatchCard(m.league, m.home, m.away, m.hP, m.aP, m.pred));
            });
        }

        // Initialize script automatically when user enters page
        fetchMatches();
    </script>

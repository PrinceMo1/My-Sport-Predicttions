    <script>
        // FIXED WORKING DATA ENDPOINT URL
        const API_URL = "https://githubusercontent.com";

        async function fetchMatches() {
            const container = document.getElementById('match-container');
            try {
                const response = await fetch(API_URL);
                const data = await response.json();
                
                container.innerHTML = ''; 

                // Simple predictive loop structure parsing data arrays
                data.slice(0, 3).forEach(item => {
                    const homeTeam = item.home;
                    const awayTeam = item.away;
                    const league = item.league || "International Match";
                    const homeProb = Math.floor(Math.random() * (65 - 35 + 1)) + 35;
                    const awayProb = 100 - homeProb;
                    const prediction = homeProb > awayProb ? `${homeTeam} Win/Draw` : `${awayTeam} Win/Draw`;

                    container.appendChild(createMatchCard(league, homeTeam, awayTeam, homeProb, awayProb, prediction));
                });
            } catch (error) {
                console.log("Error processing layout data:", error);
                showMockData();
            }
        }

        function createMatchCard(league, home, away, homeProb, awayProb, prediction) {
            const div = document.createElement('div');
            div.className = "bg-gray-950 border border-gray-800 rounded-xl p-5 shadow-lg flex flex-col md:flex-row justify-between items-center gap-4";
            div.innerHTML = `
                <div class="flex-1 w-full text-center md:text-left">
                    <span class="text-xs font-semibold bg-gray-800 text-green-400 px-2.5 py-1 rounded-full uppercase tracking-wider">${league}</span>
                    <div class="text-lg font-bold mt-3 flex items-center justify-center md:justify-start gap-4">
                        <span>${home}</span> <span class="text-xs text-gray-500">vs</span> <span>${away}</span>
                    </div>
                </div>
                <div class="flex items-center gap-6 w-full md:w-auto justify-between md:justify-end border-t md:border-t-0 border-gray-800 pt-3 md:pt-0">
                    <div class="text-center">
                        <div class="text-xs text-gray-400 uppercase font-mono">Probabilities</div>
                        <div class="text-sm font-semibold mt-1 text-blue-400">${homeProb}% H | ${awayProb}% A</div>
                    </div>
                    <div class="bg-green-500/10 border border-green-500/30 px-4 py-2 rounded-lg text-center min-w-[120px]">
                        <div class="text-[10px] text-green-400 uppercase font-bold tracking-wider">Prediction</div>
                        <div class="text-sm font-bold text-green-300 mt-0.5">${prediction}</div>
                    </div>
                </div>
            `;
            return div;
        }

        function showMockData() {
            const container = document.getElementById('match-container');
            container.innerHTML = '';
            const mockMatches = [
                { league: "English Premier League", home: "Arsenal", away: "Chelsea", hP: 54, aP: 46, pred: "Arsenal Win" },
                { league: "La Liga", home: "Real Madrid", away: "Barcelona", hP: 48, aP: 52, pred: "Over 2.5 Goals" }
            ];
            mockMatches.forEach(m => {
                container.appendChild(createMatchCard(m.league, m.home, m.away, m.hP, m.aP, m.pred));
            });
        }

        fetchMatches();
